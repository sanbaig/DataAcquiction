
/************************************************************
 * Project : StretchSense_BLE_Test
 * MCU     : Nordic nRF52832
 * SDK     : nRF Connect SDK v3.3.5
 *
 * Function:
 * 1. Initialize Bluetooth Low Energy (BLE)
 * 2. Advertise the device as a BLE Peripheral
 * 3. Create a custom GATT Service
 * 4. Generate two channels of simulated sensor data
 * 5. Transmit data through BLE Notifications
 * 6. Automatically restart advertising after disconnection
 *
 * Data transmission:
 * CH1: uint16_t
 * CH2: uint16_t
 * Sampling interval: 100 ms
 * Transmission rate: 10 packets/s
 ************************************************************/

#include <errno.h>

#include <zephyr/kernel.h>
#include <zephyr/device.h>
#include <zephyr/drivers/adc.h>
#include <zephyr/drivers/gpio.h>
#include <zephyr/sys/byteorder.h>
#include <zephyr/sys/printk.h>
#include <zephyr/sys/util.h>

#include <zephyr/bluetooth/bluetooth.h>
#include <zephyr/bluetooth/conn.h>
#include <zephyr/bluetooth/gatt.h>
#include <zephyr/bluetooth/uuid.h>

#include <hal/nrf_saadc.h>

#define DEVICE_NAME "StretchSense"

/* 原理图：ADC_CH1=P0.02/AIN0，ADC_CH2=P0.03/AIN1 */
#define ADC_CHANNEL_CH1 0
#define ADC_CHANNEL_CH2 1
#define ADC_PIN_CH1 NRF_SAADC_INPUT_AIN0
#define ADC_PIN_CH2 NRF_SAADC_INPUT_AIN1

/* 原理图：EXC_CH1=P0.14，EXC_CH2=P0.15 */
#define EXC_CH1_PIN 13
#define EXC_CH2_PIN 14

#define ADC_RESOLUTION 12
#define ADC_ACQUISITION_TIME \
    ADC_ACQ_TIME(ADC_ACQ_TIME_MICROSECONDS, 40)

#define BT_UUID_SENSOR_SERVICE_VAL \
    BT_UUID_128_ENCODE(0x12345678, 0x1234, 0x5678, \
                       0x1234, 0x56789abcdef0)

#define BT_UUID_SENSOR_DATA_VAL \
    BT_UUID_128_ENCODE(0x12345679, 0x1234, 0x5678, \
                       0x1234, 0x56789abcdef0)

#define BT_UUID_SENSOR_SERVICE \
    BT_UUID_DECLARE_128(BT_UUID_SENSOR_SERVICE_VAL)

#define BT_UUID_SENSOR_DATA \
    BT_UUID_DECLARE_128(BT_UUID_SENSOR_DATA_VAL)

static bool send_enabled;
static bool ble_connected;

static const struct device *adc_dev =
    DEVICE_DT_GET(DT_NODELABEL(adc));
static const struct device *gpio0_dev =
    DEVICE_DT_GET(DT_NODELABEL(gpio0));

static uint16_t ch1;
static uint16_t ch2;

/* 初始化 EXC 输出和两个 SAADC 输入通道 */
static int sensor_io_init(void)
{
    if (!device_is_ready(adc_dev)) {
        printk("SAADC device is not ready.\n");
        return -ENODEV;
    }

    if (!device_is_ready(gpio0_dev)) {
        printk("GPIO0 device is not ready.\n");
        return -ENODEV;
    }

    int err = gpio_pin_configure(
        gpio0_dev, EXC_CH1_PIN, GPIO_OUTPUT_ACTIVE);
    if (err) {
        printk("Failed to configure EXC_CH1: %d\n", err);
        return err;
    }

    err = gpio_pin_configure(
        gpio0_dev, EXC_CH2_PIN, GPIO_OUTPUT_ACTIVE);
    if (err) {
        printk("Failed to configure EXC_CH2: %d\n", err);
        return err;
    }

    const struct adc_channel_cfg ch1_cfg = {
        .gain = ADC_GAIN_1_4,
        .reference = ADC_REF_VDD_1_4,
        .acquisition_time = ADC_ACQUISITION_TIME,
        .channel_id = ADC_CHANNEL_CH1,
        .input_positive = ADC_PIN_CH1,
    };

    const struct adc_channel_cfg ch2_cfg = {
        .gain = ADC_GAIN_1_4,
        .reference = ADC_REF_VDD_1_4,
        .acquisition_time = ADC_ACQUISITION_TIME,
        .channel_id = ADC_CHANNEL_CH2,
        .input_positive = ADC_PIN_CH2,
    };

    err = adc_channel_setup(adc_dev, &ch1_cfg);
    if (err) {
        printk("Failed to configure ADC_CH1: %d\n", err);
        return err;
    }

    err = adc_channel_setup(adc_dev, &ch2_cfg);
    if (err) {
        printk("Failed to configure ADC_CH2: %d\n", err);
        return err;
    }

    printk("Sensor I/O ready: CH1=P0.02, CH2=P0.03, "
           "EXC1=P0.14, EXC2=P0.15.\n");

    return 0;
}

/* 同一次 ADC sequence 读取两路输入 */
static int sensor_read(uint16_t *ch1_value, uint16_t *ch2_value)
{
    int16_t sample_ch1 = 0;
    int16_t sample_ch2 = 0;

    const struct adc_sequence seq_ch1 = {
        .channels = BIT(ADC_CHANNEL_CH1),
        .buffer = &sample_ch1,
        .buffer_size = sizeof(sample_ch1),
        .resolution = ADC_RESOLUTION,
    };

    int err = adc_read(adc_dev, &seq_ch1);
    if (err) {
        return err;
    }

    const struct adc_sequence seq_ch2 = {
        .channels = BIT(ADC_CHANNEL_CH2),
        .buffer = &sample_ch2,
        .buffer_size = sizeof(sample_ch2),
        .resolution = ADC_RESOLUTION,
    };

    err = adc_read(adc_dev, &seq_ch2);
    if (err) {
        return err;
    }

    *ch1_value = (uint16_t)MAX(sample_ch1, 0);
    *ch2_value = (uint16_t)MAX(sample_ch2, 0);

    return 0;
}

static const struct bt_data advertising_data[] = {
    BT_DATA_BYTES(
        BT_DATA_FLAGS,
        BT_LE_AD_GENERAL | BT_LE_AD_NO_BREDR
    ),
    BT_DATA(
        BT_DATA_NAME_COMPLETE,
        DEVICE_NAME,
        sizeof(DEVICE_NAME) - 1
    ),
};

static void advertising_start_handler(struct k_work *work);

K_WORK_DELAYABLE_DEFINE(
    advertising_start_work,
    advertising_start_handler
);

static void advertising_start_handler(struct k_work *work)
{
    ARG_UNUSED(work);

    if (ble_connected) {
        return;
    }

    int err = bt_le_adv_start(
        BT_LE_ADV_CONN_FAST_1,
        advertising_data,
        ARRAY_SIZE(advertising_data),
        NULL,
        0
    );

    if (err == 0) {
        printk("Advertising started successfully.\n");
    } else {
        printk("Advertising start failed: %d. Retrying...\n", err);
        k_work_schedule(
            &advertising_start_work,
            K_SECONDS(1)
        );
    }
}

static void ccc_changed(
    const struct bt_gatt_attr *attr,
    uint16_t value
)
{
    ARG_UNUSED(attr);

    send_enabled = (value == BT_GATT_CCC_NOTIFY);
    printk("Notification %s.\n",
           send_enabled ? "enabled" : "disabled");
}

BT_GATT_SERVICE_DEFINE(
    sensor_service,

    BT_GATT_PRIMARY_SERVICE(BT_UUID_SENSOR_SERVICE),

    BT_GATT_CHARACTERISTIC(
        BT_UUID_SENSOR_DATA,
        BT_GATT_CHRC_NOTIFY,
        BT_GATT_PERM_NONE,
        NULL,
        NULL,
        NULL
    ),

    BT_GATT_CCC(
        ccc_changed,
        BT_GATT_PERM_READ | BT_GATT_PERM_WRITE
    )
);

static void connected(struct bt_conn *conn, uint8_t err)
{
    ARG_UNUSED(conn);

    if (err) {
        printk("Connection failed: %u\n", err);
        return;
    }

    ble_connected = true;
    send_enabled = false;

    k_work_cancel_delayable(&advertising_start_work);
    printk("BLE connected.\n");
}

static void disconnected(
    struct bt_conn *conn,
    uint8_t reason
)
{
    ARG_UNUSED(conn);

    ble_connected = false;
    send_enabled = false;

    printk("BLE disconnected: %u\n", reason);

    k_work_schedule(
        &advertising_start_work,
        K_MSEC(500)
    );
}

BT_CONN_CB_DEFINE(conn_callbacks) = {
    .connected = connected,
    .disconnected = disconnected,
};

static void sensor_work_handler(struct k_work *work);

K_WORK_DELAYABLE_DEFINE(
    sensor_work,
    sensor_work_handler
);

static void sensor_work_handler(struct k_work *work)
{
    ARG_UNUSED(work);

    int err = sensor_read(&ch1, &ch2);
    if (err) {
        printk("SAADC read failed: %d\n", err);
        k_work_schedule(&sensor_work, K_MSEC(100));
        return;
    }

    uint8_t data[4];

    sys_put_le16(ch1, &data[0]);
    sys_put_le16(ch2, &data[2]);

    if (send_enabled && ble_connected) {
        err = bt_gatt_notify(
            NULL,
            &sensor_service.attrs[2],
            data,
            sizeof(data)
        );

        if (err) {
            printk("Notification error: %d\n", err);
        }
    }

    k_work_schedule(&sensor_work, K_MSEC(100));
}

int main(void)
{
    printk("\n");
    printk("================================\n");
    printk("     StretchSense BLE Sensor\n");
    printk("================================\n");

    int err = sensor_io_init();
    if (err) {
        printk("Sensor I/O initialization failed: %d\n", err);
        return 0;
    }

    err = bt_enable(NULL);
    if (err) {
        printk("Bluetooth initialization failed: %d\n", err);
        return 0;
    }

    printk("Bluetooth initialized successfully.\n");

    k_work_schedule(
        &advertising_start_work,
        K_NO_WAIT
    );

    k_work_schedule(
        &sensor_work,
        K_MSEC(100)
    );

    while (1) {
        k_sleep(K_SECONDS(1));
    }

    return 0;
}
